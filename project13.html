Project 13: Fruit Scanner

Tell apples from bananas with a neural network on the hub.

This project is a fruit scanner: put something on the base and it tells you whether it's an apple or a banana. A force sensor notices the weight, a color sensor reads the RGB, and a small neural network running on the hub itself makes the call. B on the display and a high beep for banana, A and a low beep for apple.

When the network isn't sure (confidence under 80%), the base spins very slowly under the sensor and the answer comes from the average of a dozen readings instead of three. Every analysis gets appended to a CSV file on the hub, because it turns out the SPIKE has a file system, so the data survives between runs.

The network is tiny, three inputs, four hidden neurons, one output, and it trains itself on the hub: the first run shows a T for a few seconds, saves the weights to a file, and every run after that just loads them. The inputs aren't raw RGB. Each channel is divided by the sum of the three, so the network sees color instead of brightness, and a banana close to the sensor reads the same as a banana far from it.

The first version was trained on pure red and pure yellow, and the field test took it apart. The test objects were red and yellow LEGO bricks standing in for the fruit, and the red ones kept coming out wrong or sending the base spinning. The log on the hub showed why. A real red object isn't pure red: the sensor read one as (611, 181, 240), with even more blue than green, and what actually separates red from yellow is the green share, about 0.17 against 0.33. The log also showed dark readings, intensity under 100, where the sensor was just seeing the background, and several of those had come out as "banana".

Version two trains on the colors the sensor actually measured, pulled from that log, with the pure colors kept in at a much lower weight. Readings darker than the cutoff get ignored, and if nothing bright shows up even while spinning, the display shows X and it refuses to guess. Re-running the logged red-brick readings through the new network, on the computer, every one came out apple with over 99% confidence.

The program was written with Claude's help, and its header says so, the way the syllabus asks.

# EN1 - Classificador de frutas (maca vs banana) com rede neural no SPIKE Prime
# Generated with the assistance of Claude (Anthropic), on 2026-11-06
#
# Portas:
#   F = motor pequeno (gira a base)
#   D = sensor de forca (detecta quando algo e colocado na base)
#   E = sensor de cor (le o RGB)

from hub import port, light_matrix, sound
import motor, force_sensor, color_sensor
import time, math, random

try:
    import os
except ImportError:
    import uos as os

# ---------------- CONFIGURACAO ----------------
PORTA_MOTOR = port.F
PORTA_FORCA = port.D
PORTA_COR = port.E

ARQ_PESOS = 'nn_pesos_v2.txt'   # pesos da rede (nome novo -> treina de novo 1 vez sozinho)
ARQ_LOG = 'frutas_log.csv'      # toda analise vai pra ca
RETREINAR = False               # True = ignora os pesos salvos e treina de novo

LIMIAR_FORCA = 1        # qualquer coisa acima da base vazia (1 = 0,1 N, o menor passo do sensor)
CONF_MIN = 0.80         # confianca abaixo disso = "dificil" -> gira a base
N_LEITURAS_PARADO = 3   # leituras rapidas com a base parada
N_LEITURAS_GIRO = 12    # leituras com a base girando devagar
VEL_GIRO = 40           # graus/segundo (bem devagar)
INTERVALO_MS = 150      # tempo entre leituras
I_MIN = 100             # leitura com intensidade (i) abaixo disso = escura demais, ignora

N_ENT = 3   # entradas da rede: r, g, b normalizados
N_OCU = 4   # neuronios na camada escondida


# ---------------- ARQUIVOS ----------------
def existe(nome):
    try:
        os.stat(nome)
        return True
    except OSError:
        return False


def registrar(linha):
    novo = not existe(ARQ_LOG)
    with open(ARQ_LOG, 'a') as f:
        if novo:
            f.write('t_ms,r,g,b,i,p_banana,fruta,modo,n_leituras\n')
        f.write(linha + '\n')


def salvar_pesos(pesos):
    W1, b1, W2, b2 = pesos
    nums = [v for linha in W1 for v in linha] + b1 + W2 + [b2]
    with open(ARQ_PESOS, 'w') as f:
        f.write(','.join([str(v) for v in nums]))


def carregar_pesos():
    with open(ARQ_PESOS) as f:
        nums = [float(s) for s in f.read().split(',')]
    W1 = [nums[j * N_ENT:(j + 1) * N_ENT] for j in range(N_OCU)]
    i = N_ENT * N_OCU
    b1 = nums[i:i + N_OCU]
    i += N_OCU
    W2 = nums[i:i + N_OCU]
    i += N_OCU
    b2 = nums[i]
    return W1, b1, W2, b2


# ---------------- REDE NEURAL ----------------
def sigmoid(z):
    if z < -30:
        return 0.0
    if z > 30:
        return 1.0
    return 1 / (1 + math.exp(-z))


def tanh(z):
    if z > 15:
        return 1.0
    if z < -15:
        return -1.0
    e = math.exp(2 * z)
    return (e - 1) / (e + 1)


def aleatorio():
    # numero entre -1 e 1
    return (random.randint(0, 2000) - 1000) / 1000


def normalizar(r, g, b):
    # divide pela soma -> so a "cor", nao o brilho
    s = r + g + b
    if s == 0:
        return [0.0, 0.0, 0.0]
    return [r / s, g / s, b / s]


def frente(x, pesos):
    W1, b1, W2, b2 = pesos
    h = []
    for j in range(N_OCU):
        z = b1[j]
        for k in range(N_ENT):
            z += W1[j][k] * x[k]
        h.append(tanh(z))
    z = b2
    for j in range(N_OCU):
        z += W2[j] * h[j]
    return h, sigmoid(z)   # saida = probabilidade de ser banana


def gerar_dados():
    # Cores "tipicas" medidas pelo SEU sensor (tiradas do frutas_log.csv),
    # ja normalizadas (r', g', b'):
    #   amarelo -> g' alto (~0.33)   |   vermelho -> g' baixo (~0.17), b' parecido com g'
    # + as cores puras do comeco, pra rede nao estranhar extremos
    base = [
        ((0.44, 0.33, 0.23), 1),   # amarelo real (banana)
        ((0.59, 0.175, 0.235), 0), # vermelho real (maca)
        ((0.50, 0.50, 0.00), 1),   # amarelo puro
        ((1.00, 0.00, 0.00), 0),   # vermelho puro
    ]
    dados = []
    for (r, g, b), y in base:
        n = 25 if b > 0 else 5     # mais peso pras cores reais
        for _ in range(n):
            rr = max(0.0, r + random.randint(-30, 30) / 1000)
            gg = max(0.0, g + random.randint(-30, 30) / 1000)
            bb = max(0.0, b + random.randint(-30, 30) / 1000)
            s = rr + gg + bb
            dados.append(([rr / s, gg / s, bb / s], y))
    return dados


def treinar(dados, epocas=200, lr=0.5):
    W1 = [[aleatorio() for k in range(N_ENT)] for j in range(N_OCU)]
    b1 = [0.0] * N_OCU
    W2 = [aleatorio() for j in range(N_OCU)]
    b2 = 0.0
    for ep in range(epocas):
        for x, y in dados:
            h, p = frente(x, (W1, b1, W2, b2))
            d2 = p - y                      # erro na saida
            for j in range(N_OCU):
                d1 = d2 * W2[j] * (1 - h[j] * h[j])   # erro no neuronio escondido j
                W2[j] -= lr * d2 * h[j]
                for k in range(N_ENT):
                    W1[j][k] -= lr * d1 * x[k]
                b1[j] -= lr * d1
            b2 -= lr * d2
    return W1, b1, W2, b2


# ---------------- SENSORES ----------------
def media_leituras(n, girando):
    # So conta leituras claras o bastante (i >= I_MIN).
    # Leitura escura = sensor vendo o fundo/longe do objeto -> e so ruido.
    soma = [0, 0, 0, 0]
    validas = 0
    if girando:
        motor.run(PORTA_MOTOR, VEL_GIRO)
    for _ in range(n):
        leitura = color_sensor.rgbi(PORTA_COR)
        if leitura[3] >= I_MIN:
            for k in range(4):
                soma[k] += leitura[k]
            validas += 1
        time.sleep_ms(INTERVALO_MS)
    if girando:
        motor.stop(PORTA_MOTOR)
    if validas == 0:
        return None, 0
    return [s / validas for s in soma], validas


def prob_banana(rgbi, pesos):
    _, p = frente(normalizar(rgbi[0], rgbi[1], rgbi[2]), pesos)
    return p


def tem_algo(base_vazia):
    return force_sensor.force(PORTA_FORCA) >= base_vazia + LIMIAR_FORCA


# ---------------- PROGRAMA ----------------
def main():
    # 1) Pesos: carrega do arquivo ou treina uma vez e salva
    if existe(ARQ_PESOS) and not RETREINAR:
        pesos = carregar_pesos()
        print('Pesos carregados de', ARQ_PESOS)
    else:
        light_matrix.write('T')
        print('Treinando...')
        pesos = treinar(gerar_dados())
        salvar_pesos(pesos)
        print('Treinado e salvo em', ARQ_PESOS)

    # 2) Mede a forca com a base vazia (referencia)
    time.sleep_ms(500)
    base_vazia = force_sensor.force(PORTA_FORCA)
    print('Pronto. Forca com base vazia:', base_vazia)
    light_matrix.write('?')

    while True:
        # 3) Espera algo ser colocado (so olha a forca, sem rede neural)
        while not tem_algo(base_vazia):
            time.sleep_ms(100)
        time.sleep_ms(400)   # deixa a fruta assentar

        # 4) Leitura rapida com a base parada
        rgbi, n = media_leituras(N_LEITURAS_PARADO, False)
        modo = 'parado'
        p = prob_banana(rgbi, pesos) if rgbi else 0.5

        # 5) Se ficou escuro ou em duvida, gira devagar e tira a media
        if rgbi is None or max(p, 1 - p) < CONF_MIN:
            rgbi2, n2 = media_leituras(N_LEITURAS_GIRO, True)
            if rgbi2:
                rgbi, n = rgbi2, n2
                p = prob_banana(rgbi, pesos)
                modo = 'girando'

        if rgbi is None:
            # nenhuma leitura clara nem girando -> nao chuta
            fruta = 'nada'
            p = -1
            rgbi = [0, 0, 0, 0]
            light_matrix.write('X')
            sound.beep(200, 400, 100)
        else:
            fruta = 'banana' if p >= 0.5 else 'maca'
            light_matrix.write('B' if fruta == 'banana' else 'A')
            sound.beep(880 if fruta == 'banana' else 440, 200, 100)

        linha = '%d,%d,%d,%d,%d,%.3f,%s,%s,%d' % (
            time.ticks_ms(), rgbi[0], rgbi[1], rgbi[2], rgbi[3], p, fruta, modo, n)
        registrar(linha)
        print(linha)

        # 6) Espera tirarem a fruta antes de analisar a proxima
        while tem_algo(base_vazia):
            time.sleep_ms(100)
        light_matrix.write('?')


main()
